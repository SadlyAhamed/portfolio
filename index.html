@Service
@RequiredArgsConstructor
@Transactional
public class LoanProcessingService {

    private final LoanApplicationRepository loanRepo;
    private final ElasticsearchTemplate elasticTemplate;
    private final RabbitTemplate rabbitTemplate;
    private final AiRiskAssessmentAgent aiAgent;

    public CompletableFuture<LoanApprovalResponse> processRetailLoan(@Valid LoanRequest request) {
        // 1. High-throughput validation & persist state with Spring Data JPA
        var loan = loanRepo.save(LoanMapper.toEntity(request));

        // 2. Index real-time searchable record into Elasticsearch cluster
        elasticTemplate.save(new LoanSearchDocument(loan.getId(), loan.getCitizenId(), loan.getAmount()));

        // 3. Dispatch async event to RabbitMQ approval queue
        rabbitTemplate.convertAndSend("loan.exchange", "loan.approval.routingKey", loan);

        // 4. Run AI Agentic RAG credit risk analysis
        return aiAgent.evaluateRiskScoreAsync(loan)
            .thenApply(score -> new LoanApprovalResponse(loan.getId(), Status.PROCESSING, score));
    }
}

@Component
@RequiredArgsConstructor
public class ElasticSearchOptimizer {

    private final ElasticsearchClient esClient;

    /**
     * Sub-50ms search query across millions of enterprise banking audit records
     */
    public SearchResponse<AuditLogDoc> searchHighVolumeLogs(String keyword, DateRange range) throws IOException {
        return esClient.search(s -> s
            .index("banking-audit-logs")
            .query(q -> q
                .bool(b -> b
                    .must(m -> m.multiMatch(mm -> mm
                        .query(keyword)
                        .fields("accountNumber^3", "transactionType^2", "payload")
                        .fuzziness("AUTO")))
                    .filter(f -> f.range(r -> r.field("@timestamp")
                        .gte(JsonData.of(range.from()))
                        .lte(JsonData.of(range.to()))))
                )
            )
            .size(100),
            AuditLogDoc.class
        );
    }
}

@Component
@Slf4j
public class FinancialEventStreamProducer {

    private final KafkaTemplate<String, FinancialTransactionEvent> kafkaTemplate;

    public void streamTransaction(FinancialTransactionEvent event) {
        ProducerRecord<String, FinancialTransactionEvent> record = 
            new ProducerRecord<>("financial.transactions.v1", event.getTenantId(), event);

        kafkaTemplate.send(record).whenComplete((result, ex) -> {
            if (ex != null) {
                log.error("Failed to stream financial event: {}", event.getTxId(), ex);
            } else {
                log.info("Successfully streamed partition: {}, offset: {}",
                    result.getRecordMetadata().partition(),
                    result.getRecordMetadata().offset());
            }
        });
    }
}
